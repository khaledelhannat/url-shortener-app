# =============================================================================
# KUBERNETES LAB SAFE SHUTDOWN / STARTUP PROCEDURE
# Multi-VM Cluster + Longhorn + PostgreSQL
# =============================================================================


# =============================================================================
# BEFORE SHUTDOWN (GRACEFUL SHUTDOWN)
# =============================================================================

# -----------------------------------------------------------------------------
# 1. Check cluster health first
# -----------------------------------------------------------------------------

kubectl get nodes
kubectl get pods -A
kubectl get pvc -A


# -----------------------------------------------------------------------------
# 2. (OPTIONAL BUT RECOMMENDED)
# Scale down stateless workloads cleanly
# -----------------------------------------------------------------------------

kubectl scale deployment url-shortener-api-deployment \
--replicas=0 \
-n url-shortener-api

kubectl scale deployment redis-deployment \
--replicas=0 \
-n url-shortener-api


# -----------------------------------------------------------------------------
# 3. Verify pods terminated
# -----------------------------------------------------------------------------

kubectl get pods -n url-shortener-api


# -----------------------------------------------------------------------------
# 4. Shutdown WORKER nodes FIRST
# Run INSIDE each worker VM
# -----------------------------------------------------------------------------

sudo shutdown -h now

# OR

sudo poweroff


# -----------------------------------------------------------------------------
# 5. Shutdown CONTROL-PLANE LAST
# Run INSIDE control-plane VM
# -----------------------------------------------------------------------------

sudo shutdown -h now

# OR

sudo poweroff


# -----------------------------------------------------------------------------
# 6. AFTER all VMs are OFF
# Now safely close VMware Workstation
# -----------------------------------------------------------------------------





# =============================================================================
# AFTER POWERING ON (SAFE STARTUP)
# =============================================================================

# -----------------------------------------------------------------------------
# 1. Start CONTROL-PLANE VM FIRST
# -----------------------------------------------------------------------------

# Wait 1-2 minutes


# -----------------------------------------------------------------------------
# 2. Verify Kubernetes API is responding
# -----------------------------------------------------------------------------

kubectl get nodes


# -----------------------------------------------------------------------------
# 3. Start WORKER nodes
# -----------------------------------------------------------------------------

# Power on worker1
# Power on worker2
# Power on worker3 (if exists)


# -----------------------------------------------------------------------------
# 4. Wait for cluster recovery
# Longhorn + CSI + volumes may take time
# -----------------------------------------------------------------------------

kubectl get nodes -w


# -----------------------------------------------------------------------------
# 5. Verify all nodes become Ready
# -----------------------------------------------------------------------------

kubectl get nodes


# -----------------------------------------------------------------------------
# 6. Verify system pods
# -----------------------------------------------------------------------------

kubectl get pods -A


# -----------------------------------------------------------------------------
# 7. Verify Longhorn volumes / PVCs
# -----------------------------------------------------------------------------

kubectl get pvc -A


# -----------------------------------------------------------------------------
# 8. Scale workloads back up
# -----------------------------------------------------------------------------

kubectl scale deployment url-shortener-api-deployment \
--replicas=3 \
-n url-shortener-api

kubectl scale deployment redis-deployment \
--replicas=1 \
-n url-shortener-api


# -----------------------------------------------------------------------------
# 9. Verify application recovery
# -----------------------------------------------------------------------------

kubectl get pods -n url-shortener-api


# -----------------------------------------------------------------------------
# 10. Optional smoke verification
# -----------------------------------------------------------------------------

python smoke_runner.py


# =============================================================================
# QUICK HEALTH CHECKS
# =============================================================================

kubectl get nodes

kubectl get pods -A

kubectl get pvc -A

kubectl get svc -A

kubectl get deployments -A

kubectl get statefulsets -A
